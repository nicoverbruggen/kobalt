cmake_minimum_required(VERSION 3.22)
project(KobaltAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Network Sensors Sql)
qt_standard_project_setup(REQUIRES 6.11)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the prepared Android ARM64 SDK and the android preset")
endif()
get_filename_component(KOBALT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(retired KOBALT_DEPENDENCIES KOBALT_ASSETS)
    unset(${retired} CACHE)
endforeach()
set(source "${KOBALT_ROOT}/src")
include("${KOBALT_ROOT}/dependencies/cache.cmake")
kobalt_dependencies()
add_subdirectory("${KOBALT_ROOT}/assets" "${CMAKE_CURRENT_BINARY_DIR}/assets")


file(READ "${KOBALT_ASSETS}/manifest.json" assets)
string(JSON KOBALT_VERSION GET "${assets}" version)
option(KOBALT_ANDROID_RELEASE "Build an unsigned release APK for external signing" OFF)
if(NOT KOBALT_ANDROID_RELEASE AND NOT KOBALT_VERSION MATCHES "-debug$")
    string(APPEND KOBALT_VERSION "-debug")
endif()
string(JSON assets SET "${assets}" version "\"${KOBALT_VERSION}\"")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/asset-manifest.json" CONTENT "${assets}" @ONLY)
string(JSON count LENGTH "${assets}" files)
math(EXPR last "${count} - 1")
set(bundled)
foreach(index RANGE ${last})
    string(JSON relative GET "${assets}" files ${index} path)
    if(IS_ABSOLUTE "${relative}" OR relative MATCHES "(^|/)\\.\\.(/|$)")
        message(FATAL_ERROR "Invalid runtime asset path: ${relative}")
    endif()
    string(JSON expected GET "${assets}" files ${index} sha256)
    set(asset "${KOBALT_ASSETS}/${relative}")
    file(SHA256 "${asset}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Prepared runtime asset changed: ${relative}")
    endif()
    set_source_files_properties("${asset}" PROPERTIES QT_RESOURCE_ALIAS "${relative}")
    list(APPEND bundled "${asset}")
endforeach()

file(READ "${CRENGINE_SOURCE}/cr3gui/data/html5.css" READER_HTML_CSS)
configure_file("${source}/engine/crengine/reader_styles.h.in" reader_styles.h @ONLY)
file(GLOB patterns RELATIVE "${CRENGINE_SOURCE}/cr3gui/data/hyph" "${CRENGINE_SOURCE}/cr3gui/data/hyph/*.pattern")
list(SORT patterns)
list(TRANSFORM patterns PREPEND "\"")
list(TRANSFORM patterns APPEND "\"")
list(JOIN patterns ", " READER_HYPHENATION_PATTERNS)
configure_file("${source}/engine/crengine/reader_hyphenation.h.in" reader_hyphenation.h @ONLY)
set(KOBALT_COMMIT "" CACHE STRING "Source commit, passed by the container launcher")
option(KOBALT_COMMIT_MODIFIED "Source has uncommitted changes" OFF)
configure_file("${source}/support/build_info.h.in" build_info.h @ONLY)

file(GLOB_RECURSE shared CONFIGURE_DEPENDS "${source}/*.cpp" "${source}/*.h")
# The scenario player is built only into evaluation builds.
list(FILTER shared EXCLUDE REGEX "/src/scenario/")
list(REMOVE_ITEM shared "${source}/publication/loopback.cpp" "${source}/assets/system_fonts.cpp")
file(GLOB adapters CONFIGURE_DEPENDS "*.cpp" "*.h")
qt_add_executable(kobalt_android ${shared} ${adapters})
option(KOBALT_SQLITE_ANNOTATIONS_DEFAULT "Use SQLite annotation storage by default" ON)
if(KOBALT_SQLITE_ANNOTATIONS_DEFAULT)
    set_source_files_properties("${source}/storage/annotation_database.cpp" PROPERTIES
        COMPILE_DEFINITIONS KOBALT_SQLITE_ANNOTATIONS_DEFAULT=1)
else()
    set_source_files_properties("${source}/storage/annotation_database.cpp" PROPERTIES
        COMPILE_DEFINITIONS KOBALT_SQLITE_ANNOTATIONS_DEFAULT=0)
endif()
target_include_directories(kobalt_android PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${source}/interfaces/reader/ui/controls")
target_include_directories(kobalt_android SYSTEM PRIVATE "${CRENGINE_SOURCE}/crengine/include" "${CRENGINE_BUILD}"
    "${prefix}/include" "${prefix}/include/harfbuzz" "${prefix}/include/freetype2" "${prefix}/include/fribidi")
target_compile_options(kobalt_android PRIVATE -Wall -Wextra "-iquote${KOBALT_ROOT}" "-iquote${source}")
target_compile_definitions(kobalt_android PRIVATE KOBALT_READER_HARDWARE KOBALT_VERSION="${KOBALT_VERSION}")
include("${KOBALT_ROOT}/platforms/cache-policy.cmake")
target_compile_definitions(kobalt_android PRIVATE KOBALT_LAYOUT_POLICY="${layout_policy}")
set(engine "${CMAKE_CURRENT_BINARY_DIR}/libcrengine-reader.a")
add_custom_command(OUTPUT "${engine}"
    COMMAND "${CMAKE_OBJCOPY}" --redefine-sym FT_Load_Glyph=kobalt_font_load_glyph
        --redefine-sym FT_Render_Glyph=kobalt_font_render_glyph "${CRENGINE_BUILD}/libcrengine.a" "${engine}"
    DEPENDS "${CRENGINE_BUILD}/libcrengine.a" VERBATIM)
add_custom_target(reader-engine DEPENDS "${engine}")
add_dependencies(kobalt_android reader-engine)
target_link_libraries(kobalt_android PRIVATE Qt6::Widgets Qt6::Network Qt6::Sensors Qt6::Sql "${engine}"
    "${prefix}/lib/libharfbuzz.a" "${prefix}/lib/libxxhash.a" "${prefix}/lib/libutf8proc.a"
    "${prefix}/lib/libjpeg.a" "${prefix}/lib/liblunasvg.a" "${prefix}/lib/libfreetype.a"
    "${prefix}/lib/libfribidi.a" "${prefix}/lib/libunibreak.a" "${prefix}/lib/libpng16.a" z)
target_link_options(kobalt_android PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
set_target_properties(kobalt_android PROPERTIES QT_ANDROID_EXTRA_LIBS "${openssl}/libcrypto_3.so;${openssl}/libssl_3.so")
qt_add_resources(kobalt_android android_assets BIG_RESOURCES PREFIX "/android-assets" FILES ${bundled})
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/asset-manifest.json" PROPERTIES QT_RESOURCE_ALIAS manifest.json)
qt_add_resources(kobalt_android android_manifest PREFIX "/android-assets" FILES "${CMAKE_CURRENT_BINARY_DIR}/asset-manifest.json")
set(resources "${KOBALT_ROOT}/assets")
file(GLOB notices CONFIGURE_DEPENDS "${resources}/notices/*")
qt_add_resources(kobalt_android notices PREFIX "/kobalt" BASE "${resources}" FILES "${resources}/acknowledgments.json" ${notices}
    "${resources}/samples/words-and-letters.html" "${resources}/samples/layout.html"
    "${resources}/samples/hanging-punctuation.html")

if(KOBALT_ANDROID_RELEASE)
    set(KOBALT_ANDROID_PACKAGE com.nicoverbruggen.kobalt)
else()
    set(KOBALT_ANDROID_PACKAGE com.nicoverbruggen.kobalt.debug)
endif()
set(KOBALT_ANDROID_VERSION_CODE "1" CACHE STRING "Android update number")
if(NOT KOBALT_ANDROID_VERSION_CODE MATCHES "^[1-9][0-9]*$" OR KOBALT_ANDROID_VERSION_CODE GREATER 2100000000)
    message(FATAL_ERROR "KOBALT_ANDROID_VERSION_CODE must be between 1 and 2100000000")
endif()
set_target_properties(kobalt_android PROPERTIES QT_ANDROID_APP_NAME Kobalt QT_ANDROID_PACKAGE_NAME "${KOBALT_ANDROID_PACKAGE}"
    QT_ANDROID_VERSION_CODE "${KOBALT_ANDROID_VERSION_CODE}" QT_ANDROID_VERSION_NAME "${KOBALT_VERSION}"
    QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 36 QT_ANDROID_COMPILE_SDK_VERSION 36)
set(package_source "${CMAKE_CURRENT_BINARY_DIR}/android-package")
file(READ "${QT6_INSTALL_PREFIX}/src/android/templates/build.gradle" qt_gradle)
configure_file(app/debug-signing.gradle.in debug-signing.gradle @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/debug-signing.gradle" signing)
file(GENERATE OUTPUT "${package_source}/build.gradle" CONTENT "${qt_gradle}\n${signing}")
configure_file(app/AndroidManifest.xml.in "${package_source}/AndroidManifest.xml" @ONLY)
configure_file("${boox_jar}" "${package_source}/libs/hiddenapibypass.jar" COPYONLY)
file(GLOB_RECURSE package_files CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/app" "app/src/*" "app/res/*")
foreach(file IN LISTS package_files)
    configure_file("app/${file}" "${package_source}/${file}" COPYONLY)
endforeach()
set_target_properties(kobalt_android PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${package_source}")
