cmake_minimum_required(VERSION 3.22)
project(KobaltAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Network Sensors)
qt_standard_project_setup()
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android Qt kit to build this target")
endif()
get_filename_component(KOBALT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SRC "${KOBALT_ROOT}/src")
set(KOBALT_QT_MAJOR 6)
include("${KOBALT_ROOT}/cmake/Strings.cmake")
add_compile_options("-iquote${SRC}" -include "${SRC}/platform/android/qt_compat.h")
file(READ "${CRENGINE_SOURCE}/cr3gui/data/html5.css" READER_HTML_CSS)
configure_file("${SRC}/engine/crengine/reader_styles.h.in" reader_styles.h @ONLY)
file(GLOB READER_HYPHENATION_PATTERNS RELATIVE "${CRENGINE_SOURCE}/cr3gui/data/hyph" "${CRENGINE_SOURCE}/cr3gui/data/hyph/*.pattern")
list(SORT READER_HYPHENATION_PATTERNS)
list(TRANSFORM READER_HYPHENATION_PATTERNS PREPEND "\"")
list(TRANSFORM READER_HYPHENATION_PATTERNS APPEND "\"")
list(JOIN READER_HYPHENATION_PATTERNS ", " READER_HYPHENATION_PATTERNS)
configure_file("${SRC}/engine/crengine/reader_hyphenation.h.in" reader_hyphenation.h @ONLY)
set(READER_ENGINE "${CMAKE_CURRENT_BINARY_DIR}/libcrengine-reader.a")
add_custom_command(OUTPUT "${READER_ENGINE}"
    COMMAND "${CMAKE_OBJCOPY}" --redefine-sym FT_Load_Glyph=kobalt_font_load_glyph
        --redefine-sym FT_Render_Glyph=kobalt_font_render_glyph
        "${CRENGINE_BUILD}/libcrengine.a" "${READER_ENGINE}"
    DEPENDS "${CRENGINE_BUILD}/libcrengine.a" VERBATIM)
include("${KOBALT_ROOT}/cmake/ReaderComponents.cmake")
kobalt_publication_library(android-publication)
add_library(reader-fonts STATIC "${SRC}/assets/assets.cpp" "${SRC}/assets/bundled_fonts.cpp")
set_target_properties(reader-fonts PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(reader-fonts PUBLIC Qt6::Gui)
set(KOBALT_DEPENDENCY_PREFIX "${DEPENDENCY_PREFIX}")
add_subdirectory("${SRC}/interfaces/reader/ui" reader-widgets)
include("${KOBALT_ROOT}/cmake/EngineSources.cmake")
add_library(android-model STATIC ${KOBALT_ENGINE_SOURCES} "${READER_ENGINE}")
target_compile_definitions(android-model PRIVATE KOBALT_VERSION="${KOBALT_VERSION}")
set_target_properties(android-model PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(android-model PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(android-model SYSTEM PRIVATE "${CRENGINE_SOURCE}/crengine/include" "${CRENGINE_BUILD}"
    "${DEPENDENCY_PREFIX}/include" "${DEPENDENCY_PREFIX}/include/harfbuzz" "${DEPENDENCY_PREFIX}/include/freetype2")
target_link_libraries(android-model PUBLIC android-publication reader-fonts Qt6::Gui "${READER_ENGINE}"
    "${DEPENDENCY_PREFIX}/lib/libharfbuzz.a" "${DEPENDENCY_PREFIX}/lib/libxxhash.a"
    "${DEPENDENCY_PREFIX}/lib/libutf8proc.a" "${DEPENDENCY_PREFIX}/lib/libjpeg.a"
    "${DEPENDENCY_PREFIX}/lib/liblunasvg.a" "${DEPENDENCY_PREFIX}/lib/libfreetype.a"
    "${DEPENDENCY_PREFIX}/lib/libfribidi.a" "${DEPENDENCY_PREFIX}/lib/libunibreak.a"
    "${DEPENDENCY_PREFIX}/lib/libpng16.a" z)
include("${KOBALT_ROOT}/cmake/LayoutPolicy.cmake")
kobalt_layout_policy(LAYOUT_POLICY "${KOBALT_ROOT}" "src/publication/plain_archive.cpp")
kobalt_reading_library(android-reading android-model android-publication "${LAYOUT_POLICY}")
include("${SRC}/interfaces/reader/Sources.cmake")
configure_file("${SRC}/support/build_info.h.in" build_info.h @ONLY)
qt_add_executable(kobalt_android
    "${SRC}/platform/android/start_screen.cpp" "${SRC}/platform/android/start_screen.h"
    "${SRC}/platform/android/main.cpp" "${SRC}/platform/android/reader.cpp" "${SRC}/platform/android/reader.h"
    "${SRC}/platform/android/services.cpp" "${SRC}/platform/android/runtime.cpp"
    "${SRC}/platform/android/assets.cpp" "${SRC}/platform/android/display.cpp"
    ${KOBALT_READER_INTERFACE_SOURCES})
foreach(source platform/api/reader_orientation support/diagnostics
        publication/native_resources publication/resource_server publication/plain_archive)
    target_sources(kobalt_android PRIVATE "${SRC}/${source}.cpp")
endforeach()
target_include_directories(kobalt_android PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(kobalt_android PRIVATE KOBALT_READER_HARDWARE KOBALT_VERSION="${KOBALT_VERSION}")
target_compile_options(kobalt_android PRIVATE -Wall -Wextra)
target_link_libraries(kobalt_android PRIVATE reader-widgets android-reading Qt6::Sensors Qt6::Network)
target_link_options(kobalt_android PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
foreach(library libcrypto_3.so libssl_3.so)
    if(NOT EXISTS "${ANDROID_OPENSSL_DIR}/${library}")
        message(FATAL_ERROR "Missing Android TLS library: ${ANDROID_OPENSSL_DIR}/${library}. Run android/build.py.")
    endif()
    set_property(TARGET kobalt_android APPEND PROPERTY QT_ANDROID_EXTRA_LIBS "${ANDROID_OPENSSL_DIR}/${library}")
endforeach()
set(KOBALT_ANDROID_VERSION_CODE 1 CACHE STRING "Android update number")
if(NOT KOBALT_ANDROID_VERSION_CODE MATCHES "^[1-9][0-9]*$" OR KOBALT_ANDROID_VERSION_CODE GREATER 2100000000)
    message(FATAL_ERROR "KOBALT_ANDROID_VERSION_CODE must be between 1 and 2100000000")
endif()
set_target_properties(kobalt_android PROPERTIES QT_ANDROID_APP_NAME "Kobalt"
    QT_ANDROID_VERSION_CODE "${KOBALT_ANDROID_VERSION_CODE}"
    QT_ANDROID_VERSION_NAME "${KOBALT_VERSION}"
    QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 36 QT_ANDROID_COMPILE_SDK_VERSION 36)
file(READ "${ASSET_MANIFEST}" manifest)
string(JSON count LENGTH "${manifest}" files)
math(EXPR last "${count} - 1")
set(bundled)
foreach(index RANGE ${last})
    string(JSON name GET "${manifest}" files ${index} path)
    if(name MATCHES "^fonts/ebook-core/(.+)$")
        set(asset "${KOBALT_ROOT}/vendor/ebook-fonts/${CMAKE_MATCH_1}")
    else()
        set(asset "${KOBALT_ROOT}/resources/${name}")
    endif()
    set_source_files_properties("${asset}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
    list(APPEND bundled "${asset}")
endforeach()
# Qt's temporary resource object contains no reader code and needs no forced compatibility header.
get_directory_property(reader_compile_options COMPILE_OPTIONS)
set_property(DIRECTORY PROPERTY COMPILE_OPTIONS "")
qt_add_resources(kobalt_android android_assets BIG_RESOURCES PREFIX "/android-assets" FILES ${bundled})
set_property(DIRECTORY PROPERTY COMPILE_OPTIONS "${reader_compile_options}")
set_source_files_properties("${ASSET_MANIFEST}" PROPERTIES QT_RESOURCE_ALIAS manifest.json)
qt_add_resources(kobalt_android android_manifest PREFIX "/android-assets" FILES "${ASSET_MANIFEST}")
file(GLOB notices "${KOBALT_ROOT}/resources/notices/*")
qt_add_resources(kobalt_android notices PREFIX "/kobalt" BASE "${KOBALT_ROOT}/resources"
    FILES "${KOBALT_ROOT}/resources/acknowledgments.json" ${notices})
set(KOBALT_ANDROID_PACKAGE com.nicoverbruggen.kobalt)
set(KOBALT_ANDROID_LABEL Kobalt)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    string(APPEND KOBALT_ANDROID_PACKAGE ".debug")
    set(KOBALT_ANDROID_LABEL "Kobalt")
endif()
set_property(TARGET kobalt_android PROPERTY QT_ANDROID_APP_NAME "${KOBALT_ANDROID_LABEL}")
set(package_source "${CMAKE_CURRENT_BINARY_DIR}/android-package")
# Extend the selected Qt kit's Gradle template so its packaging settings stay current.
file(READ "${QT6_INSTALL_PREFIX}/src/android/templates/build.gradle" qt_gradle)
configure_file(app/debug-signing.gradle.in "${CMAKE_CURRENT_BINARY_DIR}/debug-signing.gradle" @ONLY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/debug-signing.gradle" debug_signing)
file(GENERATE OUTPUT "${package_source}/build.gradle" CONTENT "${qt_gradle}\n${debug_signing}")
configure_file(app/AndroidManifest.xml.in "${package_source}/AndroidManifest.xml" @ONLY)
configure_file(app/src/org/kobalt/android/KobaltActivity.java "${package_source}/src/org/kobalt/android/KobaltActivity.java" COPYONLY)
file(GLOB_RECURSE package_resources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/app" "${CMAKE_CURRENT_SOURCE_DIR}/app/res/*")
foreach(resource IN LISTS package_resources)
    configure_file("app/${resource}" "${package_source}/${resource}" COPYONLY)
endforeach()
set_target_properties(kobalt_android PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${package_source}"
    QT_ANDROID_PACKAGE_NAME "${KOBALT_ANDROID_PACKAGE}")
