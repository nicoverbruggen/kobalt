cmake_minimum_required(VERSION 3.22)
project(KobaltKindle LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(KOBALT_KINDLE_NATIVE_TESTING "Build the Kindle application with the host Qt SDK for headless checks" OFF)
if(NOT KOBALT_KINDLE_NATIVE_TESTING AND NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Use the Kindle SDK toolchain, or explicitly enable KOBALT_KINDLE_NATIVE_TESTING")
endif()
set(KOBALT_KINDLE ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Sensors Sql)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(KOBALT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(source "${KOBALT_ROOT}/src")
add_subdirectory("${KOBALT_ROOT}/dependencies" "${CMAKE_CURRENT_BINARY_DIR}/dependencies")
add_subdirectory("${KOBALT_ROOT}/assets" "${CMAKE_CURRENT_BINARY_DIR}/assets")
file(READ "${KOBALT_ASSETS}/manifest.json" assets)
string(JSON KOBALT_VERSION GET "${assets}" version)
file(READ "${CRENGINE_SOURCE}/cr3gui/data/html5.css" READER_HTML_CSS)
configure_file("${source}/engine/crengine/reader_styles.h.in" reader_styles.h @ONLY)
file(GLOB patterns RELATIVE "${CRENGINE_SOURCE}/cr3gui/data/hyph" "${CRENGINE_SOURCE}/cr3gui/data/hyph/*.pattern")
list(SORT patterns)
list(TRANSFORM patterns PREPEND "\"")
list(TRANSFORM patterns APPEND "\"")
list(JOIN patterns ", " READER_HYPHENATION_PATTERNS)
configure_file("${source}/engine/crengine/reader_hyphenation.h.in" reader_hyphenation.h @ONLY)
set(KOBALT_COMMIT "" CACHE STRING "Source commit supplied by the container launcher")
option(KOBALT_COMMIT_MODIFIED "Source has uncommitted changes" OFF)
if(NOT KOBALT_COMMIT)
    execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${KOBALT_ROOT}"
        OUTPUT_VARIABLE KOBALT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${KOBALT_ROOT}"
        OUTPUT_VARIABLE modified COMMAND_ERROR_IS_FATAL ANY)
    if(modified)
        set(KOBALT_COMMIT_MODIFIED ON)
    endif()
endif()
configure_file("${source}/support/build_info.h.in" build_info.h @ONLY)

file(GLOB_RECURSE shared CONFIGURE_DEPENDS "${source}/*.cpp" "${source}/*.h")
list(REMOVE_ITEM shared "${source}/publication/loopback.cpp")
set(adapters main.cpp window.cpp window.h services.cpp services.h display.cpp device.cpp device.h properties.cpp properties.h
    ../android/reader.cpp ../android/reader.h ../android/start_screen.cpp ../android/start_screen.h)
list(TRANSFORM adapters PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/")
qt_add_executable(kobalt_kindle ${shared} ${adapters})
set_target_properties(kobalt_kindle PROPERTIES OUTPUT_NAME kobalt RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(kobalt_kindle PRIVATE "${KOBALT_ROOT}" "${source}" "${CMAKE_CURRENT_BINARY_DIR}"
    "${source}/interfaces/reader/ui/controls")
target_include_directories(kobalt_kindle SYSTEM PRIVATE "${CRENGINE_SOURCE}/crengine/include" "${CRENGINE_BUILD}"
    "${prefix}/include" "${prefix}/include/harfbuzz" "${prefix}/include/freetype2" "${prefix}/include/fribidi"
    "${KOBALT_TEST_PNG_INCLUDE}")
target_compile_options(kobalt_kindle PRIVATE -Wall -Wextra)
include("${KOBALT_ROOT}/platforms/cache-policy.cmake")
target_compile_definitions(kobalt_kindle PRIVATE KOBALT_READER_HARDWARE KOBALT_VERSION="${KOBALT_VERSION}"
    KOBALT_LAYOUT_POLICY="${layout_policy}")
if(APPLE)
    set(engine "${CRENGINE_BUILD}/libcrengine.a")
    target_compile_definitions(kobalt_kindle PRIVATE fdatasync=fsync)
    target_link_options(kobalt_kindle PRIVATE "LINKER:-force_load,${prefix}/lib/libfreetype.a")
else()
    set(engine "${CMAKE_CURRENT_BINARY_DIR}/libcrengine-reader.a")
    add_custom_command(OUTPUT "${engine}"
        COMMAND "${CMAKE_OBJCOPY}" --redefine-sym FT_Load_Glyph=kobalt_font_load_glyph
            --redefine-sym FT_Render_Glyph=kobalt_font_render_glyph "${CRENGINE_BUILD}/libcrengine.a" "${engine}"
        DEPENDS "${CRENGINE_BUILD}/libcrengine.a" VERBATIM)
    add_custom_target(kindle-engine DEPENDS "${engine}")
    add_dependencies(kindle-engine kobalt-dependencies)
    add_dependencies(kobalt_kindle kindle-engine)
    target_link_options(kobalt_kindle PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL
        "LINKER:--whole-archive,${prefix}/lib/libfreetype.a,--no-whole-archive")
endif()
add_dependencies(kobalt_kindle kobalt-dependencies)
target_link_libraries(kobalt_kindle PRIVATE Qt6::Widgets Qt6::Network Qt6::Sensors Qt6::Sql "${engine}"
    "${prefix}/lib/libharfbuzz.a" "${prefix}/lib/libxxhash.a" "${prefix}/lib/libutf8proc.a"
    "${prefix}/lib/libjpeg.a" "${prefix}/lib/liblunasvg.a" "${prefix}/lib/libfreetype.a"
    "${prefix}/lib/libfribidi.a" "${prefix}/lib/libunibreak.a" "${KOBALT_TEST_PNG}" ZLIB::ZLIB)
if(UNIX AND NOT APPLE)
    target_link_libraries(kobalt_kindle PRIVATE dl pthread)
    set_target_properties(kobalt_kindle PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
set(resources "${KOBALT_ROOT}/assets")
file(GLOB notices CONFIGURE_DEPENDS "${resources}/notices/*")
qt_add_resources(kobalt_kindle notices PREFIX "/kobalt" BASE "${resources}"
    FILES "${resources}/acknowledgments.json" ${notices})
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/bin/assets/manifest.json"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${KOBALT_ASSETS}" "${CMAKE_BINARY_DIR}/bin/assets"
    DEPENDS ${KOBALT_ASSET_INPUTS} VERBATIM)
add_custom_target(kindle-assets DEPENDS "${CMAKE_BINARY_DIR}/bin/assets/manifest.json")
add_dependencies(kobalt_kindle kindle-assets)
add_executable(kobalt-device-info device_info.cpp device.cpp device.h properties.cpp properties.h)
target_include_directories(kobalt-device-info PRIVATE "${source}")
target_compile_options(kobalt-device-info PRIVATE -Wall -Wextra -Werror)
target_link_libraries(kobalt-device-info PRIVATE Qt6::Gui Qt6::Network)
if(UNIX AND NOT APPLE)
    set_target_properties(kobalt-device-info PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
if(NOT KOBALT_KINDLE_NATIVE_TESTING)
    add_executable(kobalt-power-bridge power_bridge.c)
    set_target_properties(kobalt-power-bridge PROPERTIES AUTOMOC OFF)
    target_compile_options(kobalt-power-bridge PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(kobalt-power-bridge PRIVATE dl)
    install(TARGETS kobalt-power-bridge RUNTIME DESTINATION bin)
    install(TARGETS kobalt-device-info RUNTIME DESTINATION bin)
    add_subdirectory(qpa)
    install(TARGETS kobalt_kindle RUNTIME DESTINATION bin)
    install(DIRECTORY "${KOBALT_ASSETS}/" DESTINATION assets)
else()
    target_sources(kobalt_kindle PRIVATE captures.cpp input_checks.cpp)
    target_compile_definitions(kobalt_kindle PRIVATE KOBALT_KINDLE_NATIVE_TESTING)
    target_compile_definitions(kobalt-device-info PRIVATE KOBALT_KINDLE_NATIVE_TESTING)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(kindle-mtk-display qpa/mtk_display_test.cpp qpa/mtk_display.cpp)
        target_link_libraries(kindle-mtk-display PRIVATE Qt6::Gui)
        target_compile_options(kindle-mtk-display PRIVATE -Wall -Wextra -Werror)
        add_test(NAME kindle.mtk-display COMMAND kindle-mtk-display)
        set_tests_properties(kindle.mtk-display PROPERTIES TIMEOUT 10)
        add_executable(kindle-refresh-queue qpa/refresh_queue_test.cpp qpa/refresh_queue.cpp qpa/refresh_queue.h)
        target_link_libraries(kindle-refresh-queue PRIVATE Qt6::Gui Threads::Threads)
        target_compile_options(kindle-refresh-queue PRIVATE -Wall -Wextra -Werror)
        add_test(NAME kindle.refresh-queue COMMAND kindle-refresh-queue)
        set_tests_properties(kindle.refresh-queue PROPERTIES TIMEOUT 15)
        if(NOT TARGET Qt6::GuiPrivate)
            find_package(Qt6 REQUIRED COMPONENTS GuiPrivate)
        endif()
        add_executable(kindle-pointer-delivery qpa/pointer_delivery_test.cpp)
        target_include_directories(kindle-pointer-delivery PRIVATE "${source}" "${KOBALT_ROOT}")
        target_link_libraries(kindle-pointer-delivery PRIVATE Qt6::Widgets Qt6::GuiPrivate)
        add_test(NAME kindle.pointer-delivery COMMAND kindle-pointer-delivery)
        set_tests_properties(kindle.pointer-delivery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal" TIMEOUT 10)
        add_executable(kindle-touch-state qpa/touch_state_test.cpp)
        target_compile_options(kindle-touch-state PRIVATE -Wall -Wextra -Werror)
        add_test(NAME kindle.touch-state COMMAND kindle-touch-state)
        add_executable(kindle-device-test device_test.cpp device.cpp device.h properties.cpp properties.h)
        target_include_directories(kindle-device-test PRIVATE "${source}")
        target_compile_definitions(kindle-device-test PRIVATE KOBALT_KINDLE_NATIVE_TESTING)
        target_compile_options(kindle-device-test PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(kindle-device-test PRIVATE Qt6::Gui)
        add_test(NAME kindle.device COMMAND "${CMAKE_COMMAND}" "-DAPP=$<TARGET_FILE:kindle-device-test>"
            "-DBUILD=${CMAKE_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/run-device-test.cmake")
        set_tests_properties(kindle.device PROPERTIES TIMEOUT 15)
        add_test(NAME kindle.launcher COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/launcher/supervisor-test.sh"
            "${CMAKE_BINARY_DIR}/test-tmp")
        add_test(NAME kindle.standalone COMMAND "${CMAKE_COMMAND}"
            "-DAPP=$<TARGET_FILE:kobalt_kindle>" "-DROOT=${KOBALT_ROOT}" "-DBUILD=${CMAKE_BINARY_DIR}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/run-smoke.cmake")
        set_tests_properties(kindle.standalone PROPERTIES TIMEOUT 90)
        add_test(NAME kindle.preview COMMAND "${CMAKE_COMMAND}"
            "-DAPP=$<TARGET_FILE:kobalt_kindle>" "-DROOT=${KOBALT_ROOT}" "-DBUILD=${CMAKE_BINARY_DIR}" -DPREVIEW=ON
            -P "${CMAKE_CURRENT_SOURCE_DIR}/run-smoke.cmake")
        set_tests_properties(kindle.preview PROPERTIES TIMEOUT 90)
    endif()
endif()
